📋 ДЕТАЛЬНОЕ ТЗ
1. Перенос файлов
1.1. Перенести все файлы из manager/src/services/collectors/ в manager/src/connectors/permissions/:

base.py → connectors/permissions/base.py

local.py → connectors/permissions/local.py

ad.py → connectors/permissions/ad.py

remote.py → connectors/permissions/remote.py

factory.py → connectors/permissions/factory.py

access_service.py → services/access_service.py

__init__.py → connectors/permissions/__init__.py

1.2. При переносе __init__.py убедиться, что он экспортирует все необходимые классы:

python
from .base import PermissionCollector
from .local import LocalPermissionCollector
from .ad import ADPermissionCollector
from .remote import RemotePermissionCollector
from .factory import PermissionCollectorFactory

__all__ = [
    'PermissionCollector',
    'LocalPermissionCollector',
    'ADPermissionCollector',
    'RemotePermissionCollector',
    'PermissionCollectorFactory',
]
1.3. В factory.py убедиться, что зарегистрированы оба коллектора:

python
_collectors = {
    "local": LocalPermissionCollector,
    "remote": RemotePermissionCollector,
    "ad": ADPermissionCollector,
}
2. Удаление дублирующей папки
2.1. Удалить всю папку manager/src/services/collectors/ после успешного переноса всех файлов.

3. Обновление импортов
3.1. Файл manager/src/services/access_service.py

Изменить импорт с:

python
from manager.src.services.collectors import PermissionCollectorFactory
на:

python
from manager.src.connectors.permissions import PermissionCollectorFactory
3.2. Файл manager/src/api/permissions.py (если создан)

Изменить импорт с:

python
from manager.src.services.collectors import PermissionCollectorFactory
на:

python
from manager.src.connectors.permissions import PermissionCollectorFactory
3.3. Файл manager/src/connectors/permissions/__init__.py

Добавить экспорт RemotePermissionCollector:

python
from .remote import RemotePermissionCollector
3.4. Проверить остальные файлы на наличие импортов из services.collectors и исправить их.
